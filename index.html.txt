<script>
    // === GANTI KODE DI BAWAH INI DENGAN firebaseConfig ANDA (DARI LANGKAH 2) ===
    const firebaseConfig = {
        apiKey: "AIzaSy...",
        authDomain: "eproject-xxxxx.firebaseapp.com",
        projectId: "eproject-xxxxx",
        storageBucket: "eproject-xxxxx.appspot.com",
        messagingSenderId: "...",
        appId: "..."
    };
    // =========================================================================

    // Inisialisasi Firebase
    firebase.initializeApp(firebaseConfig);
    const db = firebase.firestore();

    // Fungsi untuk mengambil data secara Realtime
    db.collection("opnames").orderBy("timestamp", "desc").onSnapshot((snapshot) => {
        const listContainer = document.getElementById("data-list");
        const statusText = document.getElementById("sync-status");
        
        listContainer.innerHTML = ""; // Bersihkan tampilan lama
        statusText.innerHTML = "● Terhubung (Realtime)";
        statusText.className = "status-online";

        if (snapshot.empty) {
            listContainer.innerHTML = "<p style='text-align:center'>Belum ada data opname masuk.</p>";
            return;
        }

        snapshot.forEach((doc) => {
            const data = doc.data();
            const card = `
                <div class="card">
                    <div class="info">
                        <h3>${data.itemName || 'Tanpa Nama Item'}</h3>
                        <p><b>Ruas:</b> ${data.roadName || '-'}</p>
                        <p><b>Lokasi:</b> ${data.location || '-'}</p>
                        <p><b>Tanggal:</b> ${data.dateString || '-'}</p>
                    </div>
                    <div class="volume">
                        <p>Volume Total</p>
                        <span class="vol-value">${data.calculatedVolume ? data.calculatedVolume.toFixed(3) : '0.000'} ${data.unit || ''}</span>
                    </div>
                </div>
            `;
            listContainer.innerHTML += card;
        });
    }, (error) => {
        console.error("Error: ", error);
        document.getElementById("sync-status").innerHTML = "Gagal terhubung ke database.";
    });
</script>
